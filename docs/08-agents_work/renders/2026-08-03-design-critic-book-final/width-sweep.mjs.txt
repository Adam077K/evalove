/**
 * The masthead's edge-to-edge fit has 0.3px of margin at 393. Where does it
 * break? Sweeps common viewport widths on the real /book route. Read-only.
 */
import pw from "/Users/adamks/VibeCoding/evalove/.worktrees/book-what-came-back/apps/web/node_modules/@playwright/test/index.js";
const { chromium } = pw;
import { SignJWT } from "/Users/adamks/VibeCoding/evalove/.worktrees/book-what-came-back/apps/web/node_modules/jose/dist/webapi/index.js";
import { readFileSync } from "node:fs";

const SCRATCH =
  "/private/tmp/claude-501/-Users-adamks-VibeCoding-evalove--worktrees-ceo-3-1785631504/5cf8e0ee-2e9e-4a8f-94f8-5a8a5794b97f/scratchpad";
const CREDS = JSON.parse(readFileSync(`${SCRATCH}/creds.json`, "utf8"));
const BASE = "http://localhost:3111";

function keyBytes() {
  const b = atob(CREDS.secret);
  const a = new Uint8Array(b.length);
  for (let i = 0; i < b.length; i += 1) a[i] = b.charCodeAt(i);
  return a;
}
const iat = Math.floor(Date.now() / 1000);
const token = await new SignJWT({ sid: "3f1c0d2e-1111-4222-8333-444455556666", v: 1 })
  .setProtectedHeader({ alg: "HS256", typ: "JWT" })
  .setIssuedAt(iat)
  .setExpirationTime(iat + 15552000)
  .sign(keyBytes());

const WIDTHS = [
  { w: 320, h: 568, name: "iPhone SE 1 / smallest supported" },
  { w: 360, h: 800, name: "iPhone 12 mini / common Android" },
  { w: 375, h: 667, name: "iPhone SE 2-3 / iPhone 8" },
  { w: 390, h: 844, name: "iPhone 12-14" },
  { w: 393, h: 852, name: "iPhone 15/16 — primary" },
  { w: 430, h: 932, name: "iPhone Pro Max" },
  { w: 768, h: 1024, name: "tablet portrait" },
  { w: 1280, h: 900, name: "desktop" },
];

const browser = await chromium.launch();
const rows = [];
for (const v of WIDTHS) {
  const ctx = await browser.newContext({
    viewport: { width: v.w, height: v.h },
    deviceScaleFactor: 2,
    isMobile: v.w < 700,
    hasTouch: v.w < 700,
  });
  await ctx.addCookies([
    { name: "ea_session", value: token, domain: "localhost", path: "/", httpOnly: true, secure: false, sameSite: "Lax" },
  ]);
  const page = await ctx.newPage();
  await page.goto(`${BASE}/book?mode=day`, { waitUntil: "networkidle" });
  await page.waitForTimeout(2000);
  await page.addStyleTag({ content: "nextjs-portal{display:none !important}" });
  await page.waitForTimeout(200);

  const m = await page.evaluate(() => {
    const cs = (e) => getComputedStyle(e);
    const h1 = document.querySelector("main h1");
    const main = document.querySelector("main");
    const range = document.createRange();
    range.selectNodeContents(h1);
    const g = range.getBoundingClientRect();
    const lh = parseFloat(cs(h1).lineHeight);
    const measure = main.getBoundingClientRect().width - parseFloat(cs(main).paddingLeft) - parseFloat(cs(main).paddingRight);
    // does the page scroll sideways?
    const hScroll = document.documentElement.scrollWidth > document.documentElement.clientWidth;
    // pocket link vs dock
    const nav = [...document.querySelectorAll("body *")].find((e) => cs(e).position === "fixed" && e.tagName === "NAV");
    const pocket = document.querySelector('main a[href="/pocket"]');
    const pb = pocket.getBoundingClientRect();
    const nb = nav.getBoundingClientRect();
    const icon = pocket.querySelector("svg").getBoundingClientRect();
    const hit = document.elementFromPoint((icon.left + icon.right) / 2, (icon.top + icon.bottom) / 2);
    let n = hit, inLink = false;
    while (n) { if (n === pocket) { inLink = true; break; } n = n.parentElement; }
    return {
      fontSize: +parseFloat(cs(h1).fontSize).toFixed(1),
      glyphW: +g.width.toFixed(1),
      measure: +measure.toFixed(1),
      lines: Math.round(g.height / lh),
      hScroll,
      pocketTop: +pb.top.toFixed(1),
      pocketBottom: +pb.bottom.toFixed(1),
      dockTop: +nb.top.toFixed(1),
      pocketUnderDock: pb.bottom > nb.top,
      pocketTappable: inLink,
    };
  });
  rows.push({ viewport: `${v.w}x${v.h}`, name: v.name, ...m });
  await page.screenshot({ path: `${SCRATCH}/final-renders/book-day-w${v.w}.png` });
  await ctx.close();
}
console.table(rows);
console.log(JSON.stringify(rows, null, 2));
