/**
 * Composition study rig.
 *
 * Loads the real app (so globals.css, the fonts, the noise layer, the dock and
 * the mode switch are all genuine), then replaces <main>'s children with a
 * candidate composition. Nothing in apps/web is edited.
 *
 * Every candidate is styled from the real custom properties, so night is real
 * night and the radius scale is the real scale.
 *
 * node study.mjs <outdir> <candidate-id> ...
 */
import pw from "/Users/adamks/VibeCoding/evalove/.worktrees/ceo-3-1785631504/apps/web/node_modules/@playwright/test/index.js";
const { chromium } = pw;
import { SignJWT } from "/Users/adamks/VibeCoding/evalove/.worktrees/ceo-3-1785631504/apps/web/node_modules/jose/dist/webapi/index.js";
import { readFileSync, mkdirSync } from "node:fs";
import { CANDIDATES, STUDY_CSS } from "./candidates.mjs";

const SCRATCH =
  "/private/tmp/claude-501/-Users-adamks-VibeCoding-evalove--worktrees-ceo-3-1785631504/5cf8e0ee-2e9e-4a8f-94f8-5a8a5794b97f/scratchpad";
const CREDS = JSON.parse(readFileSync(`${SCRATCH}/creds.json`, "utf8"));
const BASE = process.env.BASE ?? "http://localhost:3311";

function keyBytes() {
  const binary = atob(CREDS.secret);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i += 1) bytes[i] = binary.charCodeAt(i);
  return bytes;
}

const iat = Math.floor(Date.now() / 1000);
const token = await new SignJWT({ sid: "3f1c0d2e-1111-4222-8333-444455556666", v: 1 })
  .setProtectedHeader({ alg: "HS256", typ: "JWT" })
  .setIssuedAt(iat)
  .setExpirationTime(iat + 15552000)
  .sign(keyBytes());

const [outdir, ...ids] = process.argv.slice(2);
mkdirSync(outdir, { recursive: true });

const browser = await chromium.launch();
const ctx = await browser.newContext({
  viewport: { width: 393, height: 852 },
  deviceScaleFactor: 2,
  isMobile: true,
  hasTouch: true,
});
await ctx.addCookies([
  { name: "ea_session", value: token, domain: "localhost", path: "/", httpOnly: true, secure: false, sameSite: "Lax" },
]);
const page = await ctx.newPage();

const wanted = ids.length ? ids : Object.keys(CANDIDATES);
const report = [];

for (const id of wanted) {
  const cand = CANDIDATES[id];
  if (!cand) {
    report.push({ id, error: "unknown candidate" });
    continue;
  }
  for (const mode of ["day", "night"]) {
    await page.goto(`${BASE}/home?mode=${mode}`, { waitUntil: "networkidle" });
    // Let React hydrate and settle BEFORE the DOM is replaced, or hydration
    // paints the real page back over the candidate. That already produced one
    // false render in this session.
    await page.waitForTimeout(2500);
    await page.addStyleTag({ content: "nextjs-portal{display:none !important}" });
    await page.addStyleTag({ content: STUDY_CSS });
    await page.evaluate((html) => {
      const main = document.querySelector("main");
      main.innerHTML = `<div data-study="1">${html}</div>`;
    }, cand.html);
    await page.waitForTimeout(1600);
    const ok = await page.evaluate(() => !!document.querySelector("[data-study]"));
    if (!ok) throw new Error(`candidate ${id} was overwritten by hydration`);
    await page.screenshot({ path: `${outdir}/${id}--${mode}.png` });
    const h = await page.evaluate(() => document.documentElement.scrollHeight);
    if (true) {
      await page.evaluate(() => window.scrollTo(0, 99999));
      await page.waitForTimeout(400);
      await page.screenshot({ path: `${outdir}/${id}--${mode}--end.png` });
    }
    report.push({ id, mode, scrollHeight: h });
  }
}

console.log(JSON.stringify(report, null, 2));
await browser.close();
